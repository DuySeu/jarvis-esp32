## Goal

Provide the foundation for Phase 1: a `TuyaConfig` helper that reads Tuya credentials from NVS, and guaranteed SNTP time sync (used to generate the 13-digit timestamp Tuya requires for request signing).

No actual Tuya API calls happen in this plan — this is ground work for plans 01-02 and 01-03.

Create the directory `main/tuya/` and create `main/tuya/tuya_config.h` with the following exact content:

```cpp
#ifndef TUYA_CONFIG_H
#define TUYA_CONFIG_H

#include <string>
#include "settings.h"

// NVS namespace for Tuya credentials
// Keys (all ≤15 chars per ESP-IDF NVS limit):
//   "client_id"     — Tuya Developer client ID
//   "client_secret" — Tuya Developer client secret  
//   "region"        — Tuya data center base URL (no trailing slash)
//                     Default: "openapi.tuyaus.com"

class TuyaConfig {
public:
    // Returns stored client_id or empty string if not configured
    static std::string GetClientId() {
        Settings s("tuya", false);
        return s.GetString("client_id", "");
    }

    // Returns stored client_secret or empty string if not configured
    static std::string GetClientSecret() {
        Settings s("tuya", false);
        return s.GetString("client_secret", "");
    }

    // Returns stored region or default US data center
    static std::string GetRegion() {
        Settings s("tuya", false);
        return s.GetString("region", "openapi.tuyaus.com");
    }

    // Returns true only if both client_id and client_secret are non-empty
    static bool IsConfigured() {
        return !GetClientId().empty() && !GetClientSecret().empty();
    }

    // Saves credentials to NVS (call during provisioning)
    static void Save(const std::string& client_id,
                     const std::string& client_secret,
                     const std::string& region = "openapi.tuyaus.com") {
        Settings s("tuya", true);
        s.SetString("client_id", client_id);
        s.SetString("client_secret", client_secret);
        s.SetString("region", region);
    }
};

#endif // TUYA_CONFIG_H
```
</action>

```cpp
#ifndef TUYA_SNTP_H
#define TUYA_SNTP_H

#include "esp_sntp.h"
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "esp_log.h"

static const char* TUYA_SNTP_TAG = "TuyaSNTP";

class TuyaSntp {
public:
    // Call once after WiFi is connected. Blocks until sync completes (max 10s).
    static bool Sync() {
        if (sntp_get_sync_status() == SNTP_SYNC_STATUS_COMPLETED) {
            return true; // Already synced
        }
        esp_sntp_setoperatingmode(SNTP_OPMODE_POLL);
        esp_sntp_setservername(0, "pool.ntp.org");
        esp_sntp_init();
        
        int retries = 0;
        while (sntp_get_sync_status() != SNTP_SYNC_STATUS_COMPLETED && retries < 20) {
            vTaskDelay(pdMS_TO_TICKS(500));
            retries++;
        }
        bool synced = (sntp_get_sync_status() == SNTP_SYNC_STATUS_COMPLETED);
        if (!synced) {
            ESP_LOGE(TUYA_SNTP_TAG, "SNTP sync timed out after 10s");
        }
        return synced;
    }
    
    // Returns current UTC time in milliseconds (13 digits)
    static int64_t GetTimestampMs() {
        struct timeval tv;
        gettimeofday(&tv, NULL);
        return (int64_t)tv.tv_sec * 1000LL + tv.tv_usec / 1000;
    }
};

#endif // TUYA_SNTP_H
```
</action>
