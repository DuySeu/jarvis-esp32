## Goal

Implement `TuyaAuth` — the token lifecycle manager. It:
1. Calls `GET /v1.0/token?grant_type=1` via `TuyaClient` to obtain an access token
2. Caches the token and its expiry time in memory
3. Exposes `GetValidToken()` which auto-refreshes if within 20% of TTL (i.e., 7200s × 0.8 = 5760s = refresh after 5760s)
4. Extracts and stores the `uid` from the token response (needed in Phase 2 for `GET /v1.0/users/{uid}/devices`)

The `TuyaAuth` class is a singleton, consistent with the project's `McpServer::GetInstance()` pattern.

```cpp
#ifndef TUYA_AUTH_H
#define TUYA_AUTH_H

#include <string>
#include <ctime>
#include "esp_err.h"

// TuyaAuth — Singleton managing Tuya Cloud OAuth2 token lifecycle.
//
// Token endpoint: GET /v1.0/token?grant_type=1
// Token TTL: 7200 seconds (2 hours)
// Refresh threshold: 80% of TTL = 5760 seconds
//
// Usage:
//   std::string token = TuyaAuth::GetInstance().GetValidToken();
//   if (token.empty()) { /* credentials not configured or network error */ }
//
//   std::string uid = TuyaAuth::GetInstance().GetUid();

class TuyaAuth {
public:
    static TuyaAuth& GetInstance() {
        static TuyaAuth instance;
        return instance;
    }

    // Returns a valid access token, fetching/refreshing as needed.
    // Returns empty string if:
    //   - TuyaConfig::IsConfigured() is false (credentials missing)
    //   - Network error during token fetch
    std::string GetValidToken();

    // Returns the user UID extracted from the token response.
    // Available after the first successful GetValidToken() call.
    // Used in Phase 2 for GET /v1.0/users/{uid}/devices
    const std::string& GetUid() const { return uid_; }

    // Force a fresh token fetch (clears cached token)
    esp_err_t Refresh();

private:
    TuyaAuth() = default;
    ~TuyaAuth() = default;
    TuyaAuth(const TuyaAuth&) = delete;
    TuyaAuth& operator=(const TuyaAuth&) = delete;

    // Fetch token from Tuya API and update cached fields.
    esp_err_t FetchToken();

    std::string access_token_;
    std::string uid_;
    time_t token_fetched_at_ = 0; // epoch seconds when token was fetched
    int token_expire_time_ = 7200; // TTL in seconds (from API response)

    static constexpr float kRefreshThreshold = 0.8f; // refresh at 80% of TTL
};

#endif // TUYA_AUTH_H
```

```cpp
#include "tuya_auth.h"
#include "tuya_client.h"
#include "tuya_config.h"
#include "esp_log.h"
#include "cJSON.h"
#include <ctime>

static const char* TAG = "TuyaAuth";

// Token response structure from GET /v1.0/token?grant_type=1:
// {
//   "success": true,
//   "result": {
//     "access_token": "...",
//     "expire_time": 7200,
//     "uid": "ay15...",
//     "refresh_token": "..."
//   }
// }

esp_err_t TuyaAuth::FetchToken() {
    if (!TuyaConfig::IsConfigured()) {
        ESP_LOGW(TAG, "Tuya credentials not configured in NVS");
        return ESP_ERR_INVALID_STATE;
    }

    TuyaClient client;
    std::string response;
    esp_err_t err = client.Get("/v1.0/token?grant_type=1", "", response);
    if (err != ESP_OK) {
        ESP_LOGE(TAG, "Token request failed: %s", esp_err_to_name(err));
        return err;
    }

    // Parse JSON response
    cJSON* root = cJSON_Parse(response.c_str());
    if (!root) {
        ESP_LOGE(TAG, "Failed to parse token response JSON");
        return ESP_ERR_INVALID_RESPONSE;
    }

    // Check "success" field
    cJSON* success = cJSON_GetObjectItem(root, "success");
    if (!cJSON_IsTrue(success)) {
        cJSON* msg = cJSON_GetObjectItem(root, "msg");
        ESP_LOGE(TAG, "Token API returned error: %s",
            msg && cJSON_IsString(msg) ? msg->valuestring : "unknown");
        cJSON_Delete(root);
        return ESP_FAIL;
    }

    // Extract result fields
    cJSON* result = cJSON_GetObjectItem(root, "result");
    if (!result) {
        ESP_LOGE(TAG, "Token response missing 'result' field");
        cJSON_Delete(root);
        return ESP_ERR_INVALID_RESPONSE;
    }

    cJSON* token = cJSON_GetObjectItem(result, "access_token");
    cJSON* expire = cJSON_GetObjectItem(result, "expire_time");
    cJSON* uid = cJSON_GetObjectItem(result, "uid");

    if (!token || !cJSON_IsString(token)) {
        ESP_LOGE(TAG, "Token response missing 'access_token'");
        cJSON_Delete(root);
        return ESP_ERR_INVALID_RESPONSE;
    }

    access_token_ = token->valuestring;
    token_fetched_at_ = time(nullptr);

    if (expire && cJSON_IsNumber(expire)) {
        token_expire_time_ = expire->valueint;
    }

    if (uid && cJSON_IsString(uid)) {
        uid_ = uid->valuestring;
        ESP_LOGI(TAG, "Tuya UID: %s", uid_.c_str());
    }

    ESP_LOGI(TAG, "Token obtained successfully (TTL=%ds)", token_expire_time_);
    cJSON_Delete(root);
    return ESP_OK;
}

std::string TuyaAuth::GetValidToken() {
    if (!TuyaConfig::IsConfigured()) {
        return "";
    }

    time_t now = time(nullptr);
    bool token_empty = access_token_.empty();
    bool token_expired = (now - token_fetched_at_) >= 
                         (time_t)(token_expire_time_ * kRefreshThreshold);

    if (token_empty || token_expired) {
        ESP_LOGI(TAG, "%s — fetching new token",
            token_empty ? "No token cached" : "Token near expiry");
        if (FetchToken() != ESP_OK) {
            return "";
        }
    }

    return access_token_;
}

esp_err_t TuyaAuth::Refresh() {
    access_token_.clear();
    token_fetched_at_ = 0;
    return FetchToken();
}
```
